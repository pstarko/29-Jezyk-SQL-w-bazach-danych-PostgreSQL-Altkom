/*
1. Wyświetl nazwiska i roczne dochody pracowników(wraz z commission_pct) (107 wierszy)
*/

select
	first_name,
	last_name,
	(salary + (salary * coalesce (commission_pct,0))) * 12 roczne_dochody
from employees

/*
2. Wyświetl wszystkie informacje o departamentach sortując wynik według nazw (27 wierszy)

*/

select
	*
from departments d 
order by d.department_name 

/*
3. Wyświetl listę job_id bez duplikatów(z tabeli employees) bez departamentu 20 i 50 (14 wierszy)
*/

select
	distinct job_id
from employees e 
where e.department_id not in (20, 50)

/*
4. Wyświetl employee_id, first_name, last_name, email i phone_number pracowników na
etatach AC_ACCOUNT i FI_ACCOUNT i posortuj ich według daty zatrudnienia(malejąco) (6 wierszy)
*/


select
	employee_id ,
	first_name ,
	last_name ,
	email ,
	phone_number
	from employees
where 
	job_id in ('AC_ACCOUNT', 'FI_ACCOUNT')
order by hire_date desc

/*
5. wybierz tylko employee_id, last_name i job_id z departamentów 30,40 w kolejności rosnących
zarobków (7 wierszy)

*/

select
	employee_id, 
	last_name,
	job_id
from employees e 
where
	department_id = 30
	or department_id = 40 
order by salary

/*
6. Pokaż nazwiska i płace podstawowe(salary) tych, których płace podstawowe nie mieszczą się w
przedziale 8000 do 15000 (74 wiersze)


*/

select
	last_name,
	salary
from employees e
where 
	salary not between 8000 and 15000


/*
7. wyświetl nazwisko, etat(job_id) i numer departamentu tych których nazwisko kończy się na
„an” (4 wiersze)
*/
	
select 
	last_name,
	job_id,
	department_id 
	from employees
where	
	last_name like '%an'


/*
9. wyświetl nazwiska, etaty i stawki godzinowe tych pracowników, którzy nie pracują jako AD_VP,
PU_CLERK i SA_MAN i którzy zarabiają w przedziale od 5000 do 20000. Wyniki uszereguj
według stawek godzinowych pracowników(przyjmij 20 dniowy miesiąc pracy i 8 godzinny dzień
pracy) (tylko płaca podstawowa) (50 wierszy)

*/

select
	last_name ,
	job_id ,
	salary / 20 / 8 stawka_godzinowa
from employees
where
	job_id not in ('AD_VP','PU_CLERK', 'SA_MAN')
	and salary between 5000 and 20000
order by 	
	stawka_godzinowa

/*
12. Czy są pracownicy, którzy w ciągu 5 lat spłacą mieszkanie, które kosztowało 200 000 wiedząc,
że mogą odłożyć tylko ćwiartkę ze swojej pełnej miesięcznej pensji? Pokaż ich imiona, nazwiska
pensję miesięczną i ile mogą miesięcznie odkładać (10 wierszy)

*/

select
	first_name,
	last_name ,
	salary + (salary * coalesce(commission_pct,0)) pensja,
	(salary + (salary * coalesce(commission_pct,0))) * 0.25 odkladaja_co_miesiac
from employees
where
	(salary + (salary * coalesce(commission_pct,0))) * 0.25 * 5 * 12 > 200000
