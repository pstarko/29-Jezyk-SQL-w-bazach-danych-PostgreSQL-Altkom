/*
1. Wyświetl nazwiska i stanowiska pracowników pracujących w tym samym dziale co
pracownik o nazwisku Miller (5 wierszy)
*/
select
	e.first_name ,
	e.last_name ,
	(select job_title from jobs where job_id = e.job_id ) jt
	from employees e
where
	department_id = (select department_id from employees
						where last_name = 'Miller')

/*
2. Wyświetl następujące dane o najdłużej zatrudnionym pracowniku na stanowisku
FI_ACCOUNT (1 wiersz)
*/
select
	employee_id ,
	hire_date ,
	first_name,
	last_name,
	job_id
	from employees
where
	hire_date = (select min(hire_date) from employees
					where job_id = 'FI_ACCOUNT')
	and job_id = 'FI_ACCOUNT'

/*
3. Zobacz najdłużej pracujących pracowników każdego działu. (12 wierszy)
*/
explain
select
	last_name,
	hire_date,
	department_id,
	(select department_name from departments where department_id = e.department_id ) dn
	from employees e
where
	(e.department_id , e.hire_date) in 
				(select department_id, min(hire_date) from employees
					group by department_id )
					
					
explain			
select
	last_name,
	hire_date,
	d.department_id,
	department_name dn
	from employees e join departments d on d.department_id = e.department_id
where
	(e.department_id , e.hire_date) in 
				(select department_id, min(hire_date) from employees
					group by department_id )


	/*
4. Wyświetlaj te nazwy stanowisk, w których nie zatrudniono nikogo z działu
MARKETING. Wykorzystaj podzapytanie w podzapytaniu (17 wierszy)

*/

select
	*
	from jobs j 
where
	j.job_id in (select job_id from employees
					where department_id not in 
							(select department_id from departments													
								where department_name = 'Marketing')) 	



/*
6. Wyświetl nazwiska poniższe dane pracowników związanych ze sprzedażą i mających
innego szefa niż John Singh – wykorzystaj w zapytaniu strukturę „derrived column”
(29 wierszy)

*/

select
	e.last_name ,
	e.email ,
	e.department_id ,
	(select first_name || ' ' || last_name from employees e2 where e2.employee_id = e.manager_id ) boos_name
	from employees e
where
	e.job_id in (select job_id from jobs
					where lower(job_title) like '%sales%')
	and e.manager_id != (select employee_id from employees
							where concat(first_name,' ', last_name) = 'John Singh')
	
/*
7. Użyj skorelowanego podzapytania, aby wyświetlić informacje o pracowniku
niezatrudnionym w żadnym dziale (użyj właściwej formy operatora EXISTS) (1 wiersz)
*/

select
	first_name,
	last_name
	from employees e
where
	not exists (select 1 from departments d where e.department_id = d.department_id )


/*
8. Wyświetl nazwę działu, który płaci Twoim pracownikom najwięcej pieniędzy
miesięcznie(1 wiersz)
*/
select
	d.department_name
	from employees e1 join departments d on e1.department_id = d.department_id 
group by 
	d.department_name 
having 
	sum(salary) =
				(select
					max(t1.suma)
					from (
						select 
							department_id,
							sum(salary) suma
							from employees e
						group by 
							department_id ) t1)
							

select d.department_name, sum(salary)
from employees e
join departments d on e.department_id = d.department_id
group by d.department_name
order by sum(e.salary) desc
limit 1;



select
    d.department_name,
    sum(e.salary) as suma,
    rank() over (order by sum(e.salary) desc) as pozycja
from employees e
join departments d on e.department_id = d.department_id
group by d.department_name
